cmake_minimum_required(VERSION 3.16)
project(pract2 VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Test)
add_library(graph_core src/config.cpp src/version.cpp src/repository.cpp src/features.cpp)
target_include_directories(graph_core PUBLIC src)
target_link_libraries(graph_core PUBLIC Qt6::Core Qt6::Network)
add_executable(dependencies src/main.cpp)
target_link_libraries(dependencies PRIVATE graph_core)
enable_testing()
add_executable(config_tests tests/test_config.cpp)
target_link_libraries(config_tests PRIVATE graph_core Qt6::Test)
add_test(NAME config_tests COMMAND config_tests)
foreach(test_name IN ITEMS version repository)
    add_executable(${test_name}_tests tests/test_${test_name}.cpp)
    target_link_libraries(${test_name}_tests PRIVATE graph_core Qt6::Test)
    add_test(NAME ${test_name}_tests COMMAND ${test_name}_tests)
endforeach()
