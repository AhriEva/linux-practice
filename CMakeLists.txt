cmake_minimum_required(VERSION 3.16)
project(variant18 VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Test)
add_library(shell_core src/shell.cpp src/config.cpp src/startup.cpp src/vfs.cpp
    src/basic.cpp src/options.cpp src/identity.cpp src/mutation.cpp)
target_include_directories(shell_core PUBLIC src)
target_link_libraries(shell_core PUBLIC Qt6::Core)
add_executable(emulator src/main.cpp src/terminal.cpp)
target_link_libraries(emulator PRIVATE shell_core Qt6::Widgets)
enable_testing()
add_executable(shell_tests tests/test_shell.cpp)
target_link_libraries(shell_tests PRIVATE shell_core Qt6::Test)
target_compile_definitions(shell_tests PRIVATE
    FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
add_test(NAME shell_tests COMMAND shell_tests)
add_executable(vfs_tests tests/test_vfs.cpp)
target_link_libraries(vfs_tests PRIVATE shell_core Qt6::Test)
target_compile_definitions(vfs_tests PRIVATE
    FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
add_test(NAME vfs_tests COMMAND vfs_tests)
foreach(test_name IN ITEMS mutation startup)
    add_executable(${test_name}_tests tests/test_${test_name}.cpp)
    target_link_libraries(${test_name}_tests PRIVATE shell_core Qt6::Test)
    target_compile_definitions(${test_name}_tests PRIVATE
        FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
    add_test(NAME ${test_name}_tests COMMAND ${test_name}_tests)
endforeach()
add_executable(gui_tests tests/test_gui.cpp src/terminal.cpp)
target_link_libraries(gui_tests PRIVATE shell_core Qt6::Widgets Qt6::Test)
target_compile_definitions(gui_tests PRIVATE
    FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/examples")
add_test(NAME gui_tests COMMAND gui_tests)
set_tests_properties(gui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
